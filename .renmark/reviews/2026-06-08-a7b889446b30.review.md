    26	_FRONTMATTER_RE = re.compile(r"^---\s*\n(.*?\n)---\s*\n", re.DOTALL)
    27	_KV_RE = re.compile(r"^([a-zA-Z][a-zA-Z0-9_-]*):\s*(.*)$")

    30	def parse_frontmatter(text: str) -> dict[str, Any] | None:
    31	    """Extract a top-level YAML frontmatter block as a dict. Returns None if
    32	    no frontmatter is present. Only handles the flat key:value form — that
    33	    is the entire Claude Code SKILL.md / command convention.
    34	    """
    35	    match = _FRONTMATTER_RE.match(text)
    36	    if not match:
    37	        return None
    38	    body = match.group(1)
    39	    out: dict[str, str] = {}
    40	    for raw_line in body.splitlines():
    41	        line = raw_line.rstrip()
    42	        if not line or line.startswith("#"):
    43	            continue
    44	        m = _KV_RE.match(line)
    45	        if m:
    46	            key, value = m.group(1), m.group(2).strip()
    47	            # Strip wrapping quotes if any.
    48	            if (value.startswith("'") and value.endswith("'")) or (value.startswith('"') and value.endswith('"')):
    49	                value = value[1:-1]
    50	            out[key] = value
    51	    return out

    57	def lint_skill_files(plugin_dir: Path) -> list[str]:

   121	    """Verify every commands/<name>.md has a matching skills/<name>/SKILL.md

   129	    if not skills_dir.is_dir():
   130	        return [f"plugin: missing skills/ directory at {skills_dir}"]

   159	def lint_template_rule_blocks(template_path: Path) -> list[str]:
   160	    """Verify every BEGIN:<name> has a matching END:<name>, and order is
   161	    BEGIN then END for each block (no nesting)."""
   162	    issues: list[str] = []
   163	    if not template_path.exists():
   164	        return [f"template: not found at {template_path}"]
   165	    text = template_path.read_text(encoding="utf-8")
   166	
   167	    begins = [(m.start(), m.group(1)) for m in _BEGIN_RE.finditer(text)]
   168	    ends = [(m.start(), m.group(1)) for m in _END_RE.finditer(text)]
   169	
   170	    begin_names = [name for _, name in begins]
   171	    end_names = [name for _, name in ends]
   172	
   173	    for name in begin_names:
   174	        if begin_names.count(name) > 1:
   175	            issues.append(f"template: BEGIN:{name} appears {begin_names.count(name)} times")
   176	
   177	    missing_end = sorted(set(begin_names) - set(end_names))
   178	    for name in missing_end:
   179	        issues.append(f"template: BEGIN:{name} has no matching END:{name}")
   180	
   181	    missing_begin = sorted(set(end_names) - set(begin_names))
   182	    for name in missing_begin:
   183	        issues.append(f"template: END:{name} has no matching BEGIN:{name}")
   184	
   185	    # Order check: each BEGIN must precede its END.
   186	    for name in set(begin_names) & set(end_names):
   187	        b_pos = next(pos for pos, n in begins if n == name)
   188	        e_pos = next(pos for pos, n in ends if n == name)
   189	        if b_pos > e_pos:
   190	            issues.append(f"template: END:{name} precedes BEGIN:{name}")
   191	
   192	    # De-dup
   193	    return sorted(set(issues))

   145	    """Read ONLY the metadata + ``## Summary`` section. Returns a bounded
   146	    string suitable for orchestrator output. Caps at G3 limits regardless of
   147	    what's in the file — never returns the full body.
   148	    """

   150	    if not p.exists():

   185	def read_metadata(path: Path | str) -> dict[str, Any]:
   186	    """Parse the YAML frontmatter block at the top of an artifact.
   187	
   188	    Returns an empty dict[str, Any] if the file lacks frontmatter. Does NOT pull in
   189	    a YAML library — we control the writer's format, so a tiny line parser
   190	    is sufficient and dependency-free.
   191	    """
   192	    p = Path(path)
   193	    if not p.exists():
   194	        return {}
   195	
   196	    text = p.read_text(encoding="utf-8")
   197	    if not text.startswith("---"):
   198	        return {}
   199	
   200	    end = text.find("\n---", 3)
   201	    if end == -1:
   202	        return {}
   203	
   204	    block = text[3:end].strip("\n")
   205	    result: dict[str, Any] = {}
   206	    current_list_key: str | None = None
   207	    for raw in block.splitlines():
   208	        line = raw.rstrip()
   209	        if not line:
   210	            continue
   211	        if line.startswith("  - ") and current_list_key:
   212	            result[current_list_key].append(line[4:].strip())
   213	            continue

   216	            continue
   217	        if line.endswith(":"):
   218	            key = line[:-1].strip()
   219	            result[key] = []
   220	            current_list_key = key
   221	            continue
   222	        key, _, value = line.partition(": ")
   223	        key = key.strip()
   224	        value = value.strip()
   225	        if value == "null":
   226	            result[key] = None
   227	        elif value == "[]":
   228	            result[key] = []
   229	        elif value.lower() == "true":
   230	            result[key] = True
   231	        elif value.lower() == "false":
   232	            result[key] = False
   233	        elif value.isdigit():
   234	            result[key] = int(value)
   235	        else:
   236	            result[key] = value
   237	    return result
